# Appended to %install. CMake installs the executable versioned (its VERSION
# property: vibeshine-<version>) with %{_bindir}/vibeshine as a symlink to it, but %post
# refuses a symlinked public entrypoint ("native executable is missing or
# unsafe"). Upstream's Arch hook resolves the link; the RPM scriptlet does not.
# Make %{_bindir}/vibeshine the real file.
if [ -L %{buildroot}%{_bindir}/vibeshine ]; then
  rm -f %{buildroot}%{_bindir}/vibeshine
  mv %{buildroot}%{_bindir}/vibeshine-%{build_semver} %{buildroot}%{_bindir}/vibeshine
fi
